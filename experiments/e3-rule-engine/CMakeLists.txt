cmake_minimum_required(VERSION 3.20)
project(rule_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
enable_testing()
add_executable(rule_tests tests.cpp)
add_executable(rule_bench benchmark.cpp)
foreach(target rule_tests rule_bench)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endforeach()
add_test(NAME expressions COMMAND rule_tests)
