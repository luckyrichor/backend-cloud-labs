cmake_minimum_required(VERSION 3.20)
project(rule_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
enable_testing()
add_executable(rule_eval rule_eval.cpp)
add_executable(rule_tests tests.cpp)
add_executable(rule_bench benchmark.cpp)
add_executable(rule_batch_tests batch_tests.cpp)
add_executable(rule_slot_tests slot_tests.cpp)
add_executable(rule_slot_bench slot_benchmark.cpp)
add_executable(rule_decimal_tests decimal_tests.cpp)
foreach(target rule_eval rule_decimal_tests rule_tests rule_bench rule_batch_tests rule_slot_tests rule_slot_bench)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endforeach()
add_test(NAME expressions COMMAND rule_tests)

add_test(NAME batch_differential COMMAND rule_batch_tests)

add_test(NAME variable_slots COMMAND rule_slot_tests)

add_test(NAME decimal_grammar COMMAND rule_decimal_tests)

add_test(NAME http_eval_short_circuit COMMAND rule_eval "0 && (1/0)")
set_tests_properties(http_eval_short_circuit PROPERTIES PASS_REGULAR_EXPRESSION "^0 ")
add_test(NAME http_eval_variables COMMAND rule_eval "x*2+1" x 2)
set_tests_properties(http_eval_variables PROPERTIES PASS_REGULAR_EXPRESSION "^5 ")
add_test(NAME http_eval_error COMMAND rule_eval "1/0")
set_tests_properties(http_eval_error PROPERTIES WILL_FAIL TRUE)
